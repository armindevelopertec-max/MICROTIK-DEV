creación de la carpeta-
Crea un directorio dedicado para alojar la configuración de Mikhmon:
mkdir -p ~/mikhmon-docker && cd ~/mikhmon-docker

--------------------------------------------------------------------
creación de Dockerfile

----------------------------------------------------------------------
FROM alpine:3.18

# Instalar dependencias necesarias incluyendo php81-sockets
RUN apk add --no-cache \
    nginx \
    php81 \
    php81-fpm \
    php81-curl \
    php81-session \
    php81-json \
    php81-openssl \
    php81-sockets \
    supervisor \
    curl \
    unzip

WORKDIR /var/www/html

# Descargar Mikhmon v3 desde el repositorio oficial de laksa19
RUN curl -sSL "https://github.com/laksa19/mikhmonv3/archive/refs/heads/master.zip" -o mikhmon.zip \
    && unzip -q mikhmon.zip \
    && cp -r mikhmonv3-master/* . \
    && rm -rf mikhmonv3-master mikhmon.zip \
    && chown -R nginx:nginx /var/www/html

RUN mkdir -p /run/nginx

COPY nginx.conf /etc/nginx/http.d/default.conf
COPY supervisord.conf /etc/supervisord.conf

EXPOSE 80

CMD ["/usr/bin/supervisord", "-c", "/etc/supervisord.conf"]
--------------------------------------------------------------------------------------------------------------
Crear nginx.conf
--------------------------------------------------------------------------------------------------------------
server {
    listen 80 default_server;
    listen [::]:80 default_server;
    root /var/www/html;
    index index.php index.html;

    server_name _;

    location / {
        try_files $uri $uri/ /index.php?$args;
    }

    location ~ \.php$ {
        fastcgi_pass 127.0.0.1:9000;
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;
    }

    location ~ /\.ht {
        deny all;
    }
}


-----------------------------------------------------------------------------------------------------------------
Crear supervisord.conf
------------------------------------------------------------------------------------------------------------------
[supervisord]
nodaemon=true
user=root
logfile=/var/log/supervisord.log
pidfile=/var/run/supervisord.pid

[program:php-fpm]
command=php-fpm81 -F
autostart=true
autorestart=true
stdout_logfile=/dev/stdout
stdout_logfile_maxbytes=0
stderr_logfile=/dev/stderr
stderr_logfile_maxbytes=0

[program:nginx]
command=nginx -g "daemon off;"
autostart=true
autorestart=true
stdout_logfile=/dev/stdout
stdout_logfile_maxbytes=0
stderr_logfile=/dev/stderr
stderr_logfile_maxbytes=0
----------------------------------------------------------------------------------------------------------------------
Crea el archivo docker-compose.yml:
-----------------------------------------------------------------------------------------------------------------
version: '3.8'

services:
  mikhmon:
    build: .
    container_name: mikhmon
    restart: always
    ports:
      - "80:80"
